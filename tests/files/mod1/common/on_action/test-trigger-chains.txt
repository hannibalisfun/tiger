three_year_playable_pulse = {
	trigger = {
		trigger_if = { limit = { is_adult = yes } always = yes }
		trigger_else_if = { limit = { is_adult = no } always = yes }
		trigger_else_if = { limit = { always = yes } always = yes }
		always = yes

		trigger_if = { limit = { is_adult = no } always = yes }
		trigger_else_if = {
			limit = { always = yes }
			trigger_if = { limit = { is_adult = yes } always = yes }
			trigger_else_if = { limit = { is_adult = no } always = yes }
		}
		trigger_else = { always = no }

		trigger_if = { limit = { is_adult = yes } always = yes }
	}
	effect = {
		if = { limit = { is_adult = yes } add_gold = 1 }
		else_if = { limit = { is_adult = no } add_gold = 2 }
	}
}
